Execute this in browser console (F12) at http://localhost:8000/tests/run_production_tests.html:

CalculatorTestSuite.runAllCalculatorTests().then(results => {
    console.log('✅ Test Results:', results);
    console.log(`📊 Pass Rate: ${((results.passedTests / results.totalTests) * 100).toFixed(2)}%`);
    console.log(`✅ Passed: ${results.passedTests}, ❌ Failed: ${results.failedTests}, 📈 Total: ${results.totalTests}`);
    if (results.byCategory) {
        console.log('\n📊 By Category:');
        Object.entries(results.byCategory).forEach(([cat, stats]) => {
            const rate = ((stats.passed / stats.total) * 100).toFixed(1);
            console.log(`   ${cat}: ${stats.passed}/${stats.total} (${rate}%)`);
        });
    }
}).catch(err => {
    console.error('❌ Error:', err);
});
